return function(h)
  for _,silent in ipairs({false,true}) do
    h.test("MPE layout PSET silent="..tostring(silent),function()
      local c=h.load_strata();h.finish_startup(c)
      local S=h.local_value("S")
      c.midi.plug(1,"keys")
      params:set("midi_channel_mode",4)
      params:set("v1_zone_low",72)
      local path=c.data_dir.."strata-01.pset";params:write(path,"layout")
      params:set("v1_zone_low",0)
      c.midi.send(1,{type="note_on",note=60,vel=80,ch=2})
      h.assert_true(S.lib.notes.find(1,"midi",2*128+60)~=nil)
      params:read(path,silent)
      h.assert_equal(S.lib.notes.find(1,"midi",2*128+60),nil,"prior layout's key released")
    end)
  end
end
