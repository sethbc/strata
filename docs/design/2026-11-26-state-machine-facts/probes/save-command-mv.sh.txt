#!/bin/sh
case "${STRATA_SAVE_PROBE_PATH:-}" in
  '') ;;
  *) if [ "$3" = "$STRATA_SAVE_PROBE_PATH" ] && mkdir "$3.first-mv" 2>/dev/null; then
       i=0
       while [ ! -e "$3.release" ] && [ "$i" -lt 1000 ]; do
         sleep 0.01
         i=$((i+1))
       done
     fi ;;
esac
exec /bin/mv "$@"
